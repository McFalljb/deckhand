# Resizing Atlas's drawer by its left edge (a column divider, `divider 2`),
# over Atlas's running terminal, sliding over and pinned: the width is saved,
# clamps to the minimum and to the room there is, and a double-click resets
# it. Run with tests/scripts/agent.sh tests/scripts/drawer-resize.txt.
# Screenshots: build/shots/drawer-resize/.
resize 1600 900
wait 1000
key j cmd
waitfor control token: yes 10000
assert_state layout.drawer_width 420
assert_state layout.drawer_pinned false
# ---- sliding over: the edge over Atlas's terminal drags, it does not select
divider 2 -100
wait 300
assert_state layout.drawer_width 520
assert_state home.open true
assert_state focus.id agent_input
shot drawer-resize/wider
# narrower than the minimum: 320
divider 2 400
wait 200
assert_state layout.drawer_width 320
# wider than there is room: 70% of the space right of the left column (0.7 * 1300)
divider 2 -2000
wait 200
assert_state layout.drawer_width 910
# a double-click resets it
divider 2 clicks 2
wait 200
assert_state layout.drawer_width 420
# ---- pinned: the same edge, now between the panels and the drawer
clickid drawer_pin
wait 400
assert_state layout.drawer_pinned true
divider 2 -60
wait 300
assert_state layout.drawer_width 480
shot drawer-resize/pinned
# the centre keeps its minimum: 1300 - 320 (panels) - 360 (centre)
divider 2 -2000
wait 200
assert_state layout.drawer_width 620
# at that limit the panels cannot grow into the drawer: both keep their widths
divider 1 -40
wait 200
assert_state layout.right_width 320
assert_state layout.drawer_width 620
# narrower, then wider again: back up to the limit only (presses apart: not a double-click)
wait 600
divider 1 40
wait 200
assert_state layout.right_width 280
divider 1 -100
wait 200
assert_state layout.right_width 320
assert_state layout.drawer_width 620
divider 2 clicks 2
wait 200
assert_state layout.drawer_width 420
# the panels' divider still works beside a pinned drawer
divider 1 -40
wait 200
assert_state layout.right_width 360
divider 1 clicks 2
wait 200
assert_state layout.right_width 320
# hovering the edge: the divider line, the resize cursor
divider 2 0
shot drawer-resize/hover
clickid drawer_pin
wait 400
assert_state layout.drawer_pinned false
quit
