# Which pages have a browser (tests/scripts/browser.sh; round 2): CEF has 8 slots. Only the user
# unloads a page: the tab on screen gets one, and with none free the least recently shown hidden
# page goes, its own session's first. An agent's browser_open never changes the tab on screen and
# never unloads: it loads into a free slot or waits for one. Unloaded tabs load again when shown.
wait 1500
# session A: four pages (each opened by the user's script, so each was on screen)
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
# session B: three more: 7 of 8 slots
command {"type":"new_chat","project_id":$project}
waitstate 5000 open_sessions.# 2
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
command {"type":"browser_open","url":"deckhand-test://web/nav-a.html"}
web_title Nav A 10000
web_waitpaint 10000
wait 300
# an agent's open (a control caller) in the session on screen: a slot is free, it loads behind
# the tab on screen; nothing is unloaded, in either session
control_command {"type":"browser_open","session":$session,"url":"deckhand-test://web/nav-b.html"}
waitstate 10000 open_sessions.1.tabs.4.page open
assert_state open_sessions.1.active_tab $prev
assert_state open_sessions.0.tabs.1.page open
assert_state open_sessions.0.tabs.4.page open
assert_state open_sessions.1.tabs.1.page open
# all 8 taken: the next agent tab waits and unloads nothing
control_command {"type":"browser_open","session":$session,"url":"deckhand-test://web/nav-b.html"}
wait 500
assert_state open_sessions.1.tabs.5.page waiting
web_title Nav A 1000
assert_state open_sessions.0.tabs.1.page open
assert_state open_sessions.0.tabs.2.page open
assert_state open_sessions.0.tabs.3.page open
assert_state open_sessions.0.tabs.4.page open
assert_state open_sessions.1.tabs.1.page open
assert_state open_sessions.1.tabs.2.page open
assert_state open_sessions.1.tabs.3.page open
assert_state open_sessions.1.tabs.4.page open
# the user shows it: the least recently shown hidden page of its own session goes (B's first),
# not session A's older ones
clickid tab 5
web_title Nav B 10000
web_waitpaint 10000
wait 300
assert_state open_sessions.1.tabs.5.page open
assert_state open_sessions.1.tabs.1.page unloaded
assert_state open_sessions.1.tabs.1.url deckhand-test://web/nav-a.html
assert_state open_sessions.0.tabs.1.page open
web_assert message -
# shown again, it loads (the next least recently shown hidden page of B goes)
clickid tab 1
web_title Nav A 10000
web_waitpaint 10000
wait 300
assert_state open_sessions.1.tabs.1.page open
assert_state open_sessions.1.tabs.2.page unloaded
assert_state open_sessions.0.tabs.1.page open
web_assert message -
shot browser-pages/shown-again
web_assert overwrites 0
quit
