# Terminal input: selection drawn during a drag, drags reported to an app
# with its button, macOS line-editing keys. Run with tests/scripts/run.sh.
# Screenshots go to build/shots/term/.
wait 1000
waitfor ARGS: 15000
command {"type":"new_shell_tab","session":$session}
wait 800
command {"type":"send_text","tab_id":$tab,"text":"clear; echo 'hello-world foo.bar (baz) /tmp/x_y'\n"}
wait 800
# ---- the selection is on screen while the button is still down
mousecell down 2 0
wait 100
assert_drawn_selection 0
mousecell move 8 0
wait 200
assert_drawn_selection 1
assert_selection llo-wor
mousecell move 18 0
shot term/drag-mid
assert_drawn_selection 1
assert_selection llo-world foo.bar
mousecell up 18 0
wait 200
assert_selection llo-world foo.bar
# shift-click extends, a plain click clears
mousecell down 22 0 shift
mousecell up 22 0 shift
assert_selection llo-world foo.bar (ba
mousecell down 5 0
mousecell up 5 0
wait 200
assert_selection 
assert_drawn_selection 0
# ---- a drag cut short by a tab switch (⌘1, ⌘2) or the window going to the
# background ends there: back in the tab, a move past the top edge neither
# extends the selection nor scrolls
command {"type":"send_text","tab_id":$tab,"text":"clear; seq 1 300; echo END-MARK\n"}
wait 800
mousecell down 2 3
mousecell move 2 5
key 1 cmd
wait 200
key 2 cmd
wait 200
mousecell move 2 -3
wait 800
assert_screen END-MARK
mousecell down 2 3
mousecell move 2 5
window back
window front
mousecell move 2 -3
wait 800
assert_screen END-MARK
mousecell up 2 -3
# ---- an app with button-motion mouse tracking (1002 + SGR) gets the drag
