# Browser navigation (tests/scripts/browser.sh; docs/BROWSER.md §6 step 2): browser_open, a link,
# back and forward by command and by the toolbar's buttons, the URL field, reload; State per tab
# (url, title, loading, can_back, can_forward). Fixtures from tests/web via pidesk-test://web/.
wait 1500

assert_state open_sessions.0.tabs.1.kind browser

assert_state open_sessions.0.tabs.1.can_back false
assert_state open_sessions.0.tabs.1.can_forward false
# history is refused while there is none

shot browser-nav/page-b

# the URL field: a click takes the keyboard (the page loses it); Enter goes; the page has it back

wait 100

# a text that is not an address stays in the field with its reason
clickid br_url_box

type two words

key escape
