# Run 4 of keep-sessions.sh: no holder kept anything (Stop All Sessions): the session starts
# fresh shells, in a new holder (keep_sessions is still on). The scene kills that holder (-9)
# once screen_hashes has logged: every tab ends, saying the session holder stopped; the bar's
# Restart session holder then starts a new one, which holds the next new tab.
wait 300
waitstate 8000 open_sessions.0.tabs.6.status running
waitstate 3000 open_sessions.0.tabs.6.held 7
waitfor Restored tab 5000
screen_hashes
waitstate 8000 open_sessions.0.tabs.6.holder_lost true
waitstate 3000 open_sessions.0.tabs.0.holder_lost true
assert_state open_sessions.0.tabs.6.status exited
# #108: the bar over the centre says so, and its Restart session holder starts a new one: a tab
# opened after it runs in the new holder (its terminal 1), and the ended tabs stay ended
wait 1000
hover holder_bar_go
wait 300
clickid holder_bar_go
wait 2000
assert_toast A session holder runs again
command {"type":"new_shell_tab","session":$session}
assert_result ok true
waitstate 8000 open_sessions.0.tabs.7.held 1
assert_state open_sessions.0.tabs.6.status exited
quit
