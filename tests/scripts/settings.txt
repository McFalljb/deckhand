# The Settings page (tests/scripts/settings.sh). A `wait` after anything that closes an overlay.
# Its items by `clickid spi @KEY` (settings_page_slot): a section's name, an action's id, a pi
# extension's name, or a file's path by its end.
wait 1500
assert_state open_sessions.# 1
# ---- ⌘, shows the page in the centre with the keyboard; no session on screen
key comma cmd
wait 300
assert_state focus.id settings_page
assert_menu settings_tab checked
assert_drawn Appearance
shot settings/page-dark
# ---- Coding agents (its first card); the search lists pi's files by an allowlist, never
# credentials or pi's own folders (a search: the rows stay at the top of a short window)
clickid spi @Agents
wait 500
assert_drawn Coding agents
shot settings/agents-dark
type agent/
eventually
assert_drawn models.json
assert_drawn AGENTS.md
assert_drawn prompts/
assert_not_drawn auth.json
assert_not_drawn models-store
assert_not_drawn DUMMY-AUTH-SECRET
assert_not_drawn npm/
assert_not_drawn logs/
assert_not_drawn sessions/
shot settings/search-dark
# ---- a file row's menu (right click), as the old Settings tab's: none of Rename or Trash for
# pi's own config files
rclickid spi @agent/settings.json
eventually
assert_menurows Open|Open in preview|-|Reveal in Finder|Copy path
key escape
wait 200
# ---- a folder shows its files in the search; credentials stay hidden inside it
clickid spi @agent/prompts
eventually
assert_drawn fix.md
assert_not_drawn token.json
assert_not_drawn .env

eventually

key escape
