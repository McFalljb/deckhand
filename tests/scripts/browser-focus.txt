# Keys and clicks reach the page only while it has pi-desk's keyboard
# (tests/scripts/browser.sh; review fix 1). Page: tests/web/browser.html.
wait 1500
web_open tests/web/browser.html
web_waitpaint 10000
web_title ready 5000
web_click 170 130
type ab
web_title v=ab
web_assert focused yes
# Atlas's drawer has the keyboard: typing goes there, not into the page
key j cmd
assert_state focus.id agent_input
wait 100
web_assert focused no
type zz
key backspace
web_js document.title = 'chk1 v=' + document.getElementById('field').value + ' z=' + keys.includes('KeyZ') + ' bs=' + keys.includes('Backspace')
web_title chk1 v=ab z=false bs=false
key j cmd
# the palette is open: a click on the page does not reach it
key k cmd
assert_state focus.id palette
web_click 120 50

ui_command {"type":"job_sheet","open":true}
assert_result ok true
